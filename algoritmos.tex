	A continuación se presenta el pseudocódigo estructurado con el formato estético
	oficial de un libro técnico y los colores nativos de la interfaz:
	
	\begin{algorithm}[H]
		\caption{Cálculo de Promedio y Condición de Aprobación}
		\vspace{8pt}
		\label{alg:promedio_final}
		\begin{spacing}{1.5}
			\begin{algorithmic}[1]
				
				\Proceso{promedioEstudiantes}
				\Escribir{\txt{Ingrese las tres notas del estudiante:}}
				\Leer{\texttt{nota1, nota2, nota3}}
				\Asignacion{promedio}{(nota1 + nota2 + nota3) \op{/} \num{3}}
				\Escribir{\txt{El promedio final es: }, \texttt{promedio}}
				
				\Si{\texttt{promedio} \op{$\geq$} \num{7}}
				\Escribir{\txt{El estudiante está APROBADO.}}
				\Sino
				\Escribir{\txt{El estudiante está REPROBADO.}}
				\FinSi
				
				\FinProceso
				
			\end{algorithmic}
		\end{spacing}
		\vspace{8pt}
	\end{algorithm}
