	\begin{frame}{Tema 3.- Aritmética del computador}
		
		\begin{block}{Definición:}
			La aritmética del computador estudia cómo las computadoras realizan operaciones matemáticas utilizando señales binarias representadas por 0 y 1.
		\end{block}
		
		\begin{itemize}
			\item Los computadores trabajan usando lógica digital.
			\item Todo dato se convierte en bits.
			\item Las operaciones son ejecutadas por circuitos electrónicos.
			\item La Unidad Aritmético Lógica (ALU) realiza cálculos matemáticos.
		\end{itemize}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 3.1
	% =========================================================
	\begin{frame}{Presentación}
		
		\begin{block}{Analogía}
			Una computadora es como una calculadora extremadamente rápida que sigue instrucciones exactas millones de veces por segundo.
		\end{block}
		
		\begin{itemize}
			\item No piensa como un ser humano.
			\item Solo interpreta señales eléctricas.
			\item Usa reglas matemáticas precisas.
			\item Convierte información en lenguaje binario.
		\end{itemize}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 3.2
	% =========================================================
	\begin{frame}{Aritmética del computador}
		
		\small
		
		\begin{block}{Sistema binario}
			El sistema binario utiliza únicamente dos dígitos: 0 y 1.
		\end{block}
		
		\[
		1011_2 =
		1(2^3)+0(2^2)+1(2^1)+1(2^0)=11_{10}
		\]
		
		\vspace{0.3cm}
		
		\begin{itemize}
			\item Las computadoras representan números usando bits.
			\item Cada posición binaria tiene un valor basado en potencias de 2.
			\item Las operaciones matemáticas se realizan mediante circuitos lógicos.
		\end{itemize}
		
	\end{frame}
	
	% =========================================================
	% COMPLEMENTO A DOS
	% =========================================================
	\begin{frame}{Complemento a dos}
		
		\begin{block}{Por qué se utiliza?}
			Permite representar números negativos dentro del computador.
		\end{block}
		
		\vspace{0.005cm}
		
		\begin{block}{Ejemplo}
			Representar $-5$ en 8 bits:
		\end{block}
		
		\[
		5 = 00000101
		\]
		
		Invertir bits:
		
		\[
		11111010
		\]
		
		Sumar 1:
		
		\[
		11111011
		\]
		
		\vspace{0.005cm}
		
		\begin{block}{Idea}
			Es como poner un número ``al revés'' para que el computador entienda que es negativo.
		\end{block}
		
	\end{frame}
	
	% =========================================================
	% PUNTO FLOTANTE
	% =========================================================
	\begin{frame}{Representación de punto flotante}
		
		\small
		
		\begin{block}{Explicación}
			Los números decimales se almacenan usando una estructura parecida a la notación científica.
		\end{block}
		
		\begin{itemize}
			\item Se divide en signo, exponente y mantisa.
			\item Permite representar números muy grandes o pequeños.
			\item Sigue estándares internacionales como IEEE 754.
			\item Algunos decimales no pueden representarse exactamente.
		\end{itemize}
		
		\vspace{0.3cm}
		
		\begin{block}{Ejemplo}
			Por eso operaciones como:
			
			\[
			0.1 + 0.2
			\]
			
			a veces no producen exactamente:
			
			\[
			0.3
			\]
		\end{block}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 3.3
	% =========================================================
	\begin{frame}{Unidades de medida}
		
		\small
		
		\begin{block}{Bit y byte}
			Un bit es la unidad mínima de información. Ocho bits forman un byte.
		\end{block}
		
		\vspace{0.3cm}
		
		\begin{center}
			
			\begin{tabular}{>{\bfseries}c c}
				
				\toprule
				Unidad & Equivalencia \\
				\midrule
				Bit & 0 o 1 \\
				Byte & 8 bits \\
				Kilobyte & 1024 bytes \\
				Megabyte & 1024 KB \\
				Gigabyte & 1024 MB \\
				Terabyte & 1024 GB \\
				\bottomrule
				
			\end{tabular}
			
		\end{center}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 3.4
	% =========================================================
	\begin{frame}{Lógica}
		
		\begin{block}{Lógica digital}
			La lógica digital permite que el computador tome decisiones usando verdadero o falso.
		\end{block}
		
		\begin{itemize}
			\item AND: ambas condiciones deben cumplirse.
			\item OR: basta una condición verdadera.
			\item NOT: invierte el valor.
			\item XOR: verdadero solo si las entradas son diferentes.
		\end{itemize}
		
		\vspace{0.3cm}
		
		\begin{block}{Analogía}
			AND funciona como una puerta con dos cerraduras: ambas deben abrirse.
		\end{block}
		
	\end{frame}
	
	% =========================================================
	% TEMA 4
	% =========================================================
	\begin{frame}{Tema 4.- Fundamentos de TI}
		
		\begin{block}{Definición}
			Las Tecnologías de la Información integran hardware, software, redes y datos para resolver problemas y facilitar la comunicación.
		\end{block}
		
		\begin{itemize}
			\item Automatizan procesos.
			\item Permiten almacenar información.
			\item Facilitan comunicación global.
			\item Apoyan la toma de decisiones.
		\end{itemize}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 4.1
	% =========================================================
	\begin{frame}{Presentación}
		
		\begin{block}{Analogía}
			La TI es como el sistema nervioso de una empresa: recibe información, la procesa y genera respuestas.
		\end{block}
		
		\begin{itemize}
			\item Hardware: parte física.
			\item Software: instrucciones.
			\item Redes: comunicación.
			\item Datos: información procesada.
		\end{itemize}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 4.2
	% =========================================================
	\begin{frame}{Elementos de un computador}
		
		\small
		
		\begin{center}
			
			\begin{tabular}{>{\bfseries}p{3cm} p{8cm}}
				
				\toprule
				Elemento & Función \\
				\midrule
				CPU & Ejecuta instrucciones. \\
				RAM & Guarda datos temporales. \\
				Disco SSD/HDD & Almacena información permanentemente. \\
				Teclado y mouse & Permiten entrada de datos. \\
				Monitor & Muestra resultados. \\
				Tarjeta de red & Conecta dispositivos. \\
				\bottomrule
				
			\end{tabular}
			
		\end{center}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 4.3
	% =========================================================
	\begin{frame}{Representación de la información}
		
		\begin{block}{Idea principal}
			Todo dentro del computador se representa usando bits.
		\end{block}
		
		\begin{itemize}
			\item Texto: ASCII y Unicode.
			\item Imágenes: píxeles.
			\item Audio: muestras digitales.
			\item Video: secuencia de imágenes.
		\end{itemize}
		
		\vspace{0.3cm}
		
		\begin{block}{Explicación}
			Para una computadora, una canción, una imagen y un texto son diferentes combinaciones de ceros y unos.
		\end{block}
		
	\end{frame}

			\item Internet de las cosas.

	% TEMA 5
	% =========================================================
	\begin{frame}{Tema 5.- Fundamentos de hardware}
		
		\begin{block}{Definición}
			El hardware corresponde a todos los componentes físicos de un sistema computacional.
		\end{block}
		
		\begin{itemize}
			\item Procesador.
			\item Memoria.
			\item Tarjeta madre.
			\item Dispositivos de entrada y salida.
			\item Unidades de almacenamiento.
		\end{itemize}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 5.1
	% =========================================================
	\begin{frame}{Presentación}
		
		\begin{block}{Analogía}
			El hardware funciona como una fábrica donde cada componente cumple una tarea específica.
		\end{block}
		
		\begin{itemize}
			\item CPU: director de operaciones.
			\item Memoria: espacio de trabajo.
			\item Disco: almacén.
			\item Buses: carreteras de información.
		\end{itemize}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 5.2
	% =========================================================
	\begin{frame}{Estructura y organización de un computador}
		
		\small
		
		\begin{block}{Arquitectura de Von Neumann}
			Modelo clásico utilizado en la mayoría de computadores modernos.
		\end{block}
		
		\begin{itemize}
			\item Unidad de control.
			\item Unidad aritmético lógica.
			\item Memoria principal.
			\item Entrada y salida.
		\end{itemize}
		
		\vspace{0.3cm}
		
		\begin{block}{Ciclo básico}
			Buscar \textrightarrow\ Decodificar \textrightarrow\ Ejecutar.
		\end{block}
		
	\end{frame}
	
	% =========================================================
	% SUBTEMA 5.3
	% =========================================================
	\begin{frame}{Funcionamiento interno del computador}
		
		\small
		
		\begin{enumerate}
			\item La CPU busca una instrucción.
			\item Interpreta qué debe hacer.
			\item Ejecuta la operación.
			\item Guarda el resultado.
		\end{enumerate}
		
		\vspace{0.4cm}
		
		\begin{block}{Explicación sencilla}
			La computadora sigue instrucciones paso a paso igual que una receta de cocina.
		\end{block}
		
		\vspace{0.2cm}
		
		\begin{itemize}
			\item Registros: memoria ultrarrápida.
			\item Caché: acceso rápido frecuente.
			\item Buses: transporte interno.
			\item Reloj: sincroniza operaciones.
		\end{itemize}
		
	\end{frame}
